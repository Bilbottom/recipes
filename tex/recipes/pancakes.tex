\subsection*{Ingredients}

\begin{itemize}[nolistsep]
    \item \ingredient{Plain flour}{135g (4\pfrac{3}{4}oz)}
    \item \ingredient{Baking powder}{1 tsp}
    \item \ingredient{Salt}{\pfrac{1}{2} tsp}
    \item \ingredient{Caster Sugar}{2 tbsp}
    \item \ingredient{Milk}{130ml (4\pfrac{1}{2} fl oz)}
    \item \ingredient{Large Egg}{1}
    \item \ingredient{Butter}{2 tbsp}
\end{itemize}

Add \blue{raspberries} or \blue{chocolate chips} for extra sweetness. Serve with \blue{maple syrup}.

\subsection*{Instructions}

\begin{enumerate}
    \item In a small bowl, melt the \blue{butter} and leave to cool.
    \item Sift the \blue{flour}, \blue{baking powder}, \blue{salt}, and \blue{caster sugar} into a big bowl and combine.
    \item In a separate bowl, lightly beat the \blue{egg}. Add the \blue{milk} and \blue{butter} and whisk.
    \item Pour the wet ingredients into the dry ones, mixing until smooth and thick. Let the batter stand for a few minutes.
    \item (Optional) Add the \blue{raspberries} or \blue{chocolate chips} and mix until combined.
    \item Put a small pan on a medium--low heat with a small amount of butter. When hot, add a large spoonful of batter using a ladle (or large spoon) to the pan. Cover and cook, flipping the pancake once when bubbles form on the surface.
\end{enumerate}
