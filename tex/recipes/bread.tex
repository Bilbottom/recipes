\documentclass[12pt]{article}
\usepackage{recipes}

\renewcommand{\recipename}{Bread}
\renewcommand{\recipeservings}{6 medium buns}
\renewcommand{\preptime}{70 mins}
\renewcommand{\cooktime}{25 mins}

\begin{document}
\recipetitle

\subsection*{Ingredients}

\begin{itemize}[nolistsep]
    \item \ingredient{Bread flour}{3 cups}
    \item \ingredient{Yeast}{1 sachet}
    \item \ingredient{Salt}{1 tsp}
    \item \ingredient{Warm water}{1\pfrac{1}{2} cups}
\end{itemize}

\subsection*{Instructions}

\begin{enumerate}
    \item In a large bowl, mix (don't knead!) all ingredients together. The dough should be sticky.
    \item Cover with a towel and leave to rise in a warm place for an hour.
    \item Spoon the mixture onto parchment paper, making 6 equal buns.
    \item Sprinkle on some additional \blue{flour} and, optionally, some \blue{chia seeds}.
    \item Heat the oven to 230{\degrees}C and add a water bath to the bottom shelf.
    \item Bake the dough buns for 25 minutes.
\end{enumerate}
